# The hostile corpus: inputs that try to make the sanitizer write active content.

# Every output is the reference's own output for the input above it, produced by running
# Miniflux v2.3.3 at commit c4d54f87 with the reference's test helper options: OpenLinksInNewTab
# true, base URL http://example.org/, TZ=UTC. The method is recorded in docs/testing.md. A control
# byte in an input or an output is written as an escape form, and an empty output as (empty). A data
# line is the string the reference produced for the input above it, so it is not wrapped here:
# breaking it would change what it says.

# Each rule line names the rule of the case under it, and every case here can fail on that rule: a
# case is left out when a wrong implementation of the rule it names writes the same output. So the
# corpus does not pin the raw path of a value already in the canonical encoding, a plus in a query
# that round-trips to the same string, a percent escape already canonical, an attribute allowlist
# with a single name, a read order that is already the sorted order, or the sort tie-break by value,
# which needs two attributes with the same name.

# What the corpus cannot pin: content inside math or svg, which the reference inserts through the
# foreign-content rules this port does not have. The limit of 512 open elements in parse.go is
# pinned by the two cases that mirror TestDeeplyNestedpage: the reference reaches it by panicking,
# which its own Parse recovers as an error, so a chain one past the limit answers with nothing.

# An event handler is not in the attribute allowlist for p, so only the content is written
input = <p onerror="alert(1)">x</p>
output = <p>x</p>

# A tag outside the tag allowlist writes only its content; the em inside it is in the allowlist
input = <p><span onerror="alert(1)"><em>y</em></span></p>
output = <p><em>y</em></p>

# The scheme allowlist: javascript is not in it
input = <a href="javascript:alert(1)">x</a>
output = x

# A reference without its semicolon is resolved in an attribute value, so the scheme is javascript
input = <a href="&#106;avascript&#58;alert(1)">x</a>
output = x

# data: is not in the data-URI prefix allowlist for an href
input = <a href="data:text/html,<script>alert(1)</script>">x</a>
output = x

# vbscript is not in the scheme allowlist
input = <a href="vbscript:msgbox(1)">x</a>
output = x

# A hidden element is dropped with its content
input = <div hidden>x</div>
output = (empty)

# A blocked tag is dropped with its content
input = <script>alert(1)</script>
output = (empty)

# noscript is a blocked tag, so it is dropped with its content
input = <noscript><p>x</p></noscript>
output = (empty)

# A blocked tag inside a table is dropped, and the table is written
input = <table><script>alert(1)</script></table>
output = <table></table>

# td outside a table is not an element at all, so only its content is written
input = <td><script>alert(1)</script></td>
output = (empty)

# img without a usable src has no mandatory attribute, so the element is not written at all
input = <img src="javascript:alert(1)" alt="x">
output = (empty)

# data: is not allowed in srcset, and javascript is not in the scheme allowlist
input = <img srcset="&#106;avascript:alert(1) 1x, data:text/plain;base64,SGVsbG8= 2x" alt="Example">
output = (empty)

# data:text/html is not one of the data-URI prefixes the img path allows
input = <img src="data:text/html;base64,PHNjcmlwdD5hbGVydCgxKTwvc2NyaXB0Pg==" alt="x">
output = (empty)

# A prefix in the data-URI allowlist is written as it was read, with the attribute the img path adds
input = <img src="data:image/png;base64,AAAA" alt="x">
output = <img src="data:image/png;base64,AAAA" alt="x" loading="lazy">

# A backslash is an ordinary path character for the parse, so the value is resolved as a path
input = <a href="/\\evil.com">x</a>
output = <a href="http://example.org/%5Cevil.com" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A value starting with two solidi is written as https: plus the value as read
input = <a href="//evil.example.org/path">x</a>
output = <a href="https://evil.example.org/path" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# file is not in the scheme allowlist
input = <a href="file:///etc/passwd">x</a>
output = x

# iframe is written only for a domain in its allowlist, so a javascript src drops the element
input = <iframe src="javascript:alert(1)"></iframe>
output = (empty)

# An event handler on a tag in the allowlist is dropped, and the rest of the element is written
input = <a href="https://evil.com/x" onclick="alert(1)">x</a>
output = <a href="https://evil.com/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# Content inside a kept element is written with the tags inside it that are in the allowlist
input = <p><a href="javascript:alert(1)"><b>x</b></a></p>
output = <p><b>x</b></p>

# td outside a table writes only its content, so the attribute that carries markup is not written
input = <td rowspan="<b>injection</b>">text</td>
output = text

# An entity in an attribute value is decoded, and the writer escapes the decoded ampersand again
input = <a href="https://example.org/a&amp;b">x</a>
output = <a href="https://example.org/a&amp;b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The same decoding in an img src, with the attribute the img path adds
input = <img src="https://example.org/a&amp;b" alt="x">
output = <img src="https://example.org/a&amp;b" alt="x" loading="lazy">

# Whitespace around a value is not trimmed, so the prefix check fails and the value is a path
input = <a href=" https://example.org/x ">x</a>
output = x

# div is not in the tag allowlist, so only its content is written; no depth limit is reached
input = <div><div><div><div><div><div><div><div><div><div><div><span>x</span></div></div></div></div></div></div></div></div></div></div></div>
output = x

# feed is in the scheme allowlist
input = <p><a href="feed:https://example.org/">x</a></p>
output = <p><a href="feed:https://example.org/" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a></p>

# A blocked resource substring is checked on the value as it was read, spaces included
input = <img src="https://stats.wordpress.com/b tracker.gif" alt="x">
output = (empty)

# A decoded scheme in text is not a URL, so it is written as text and the ampersand is escaped
input = <p>&#106;avascript:alert(1)</p>
output = <p>javascript:alert(1)</p>

# mailto is in the scheme allowlist
input = <a href="mailto:a@b.c">x</a>
output = <a href="mailto:a@b.c" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A decoded tab is written literally, which is the writer's rule for the six characters it knows

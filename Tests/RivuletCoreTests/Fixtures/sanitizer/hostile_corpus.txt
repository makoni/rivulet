# The hostile corpus: inputs that try to make the sanitizer write active content.

# Every output is the reference's own output for the input above it, produced by running
# Miniflux v2.3.3 at commit c4d54f87 with the reference's test helper options: OpenLinksInNewTab
# true, base URL http://example.org/, TZ=UTC. The method is recorded in docs/testing.md. A control
# byte in an input or an output is written as an escape form, and an empty output as (empty). A data
# line is the string the reference produced for the input above it, so it is not wrapped here:
# breaking it would change what it says.

# What the corpus cannot pin: content inside math or svg, which the reference inserts through the
# foreign-content rules the port does not have, and the 10000-depth guard, which belongs to the
# tokenizer and is pinned in Tests/RivuletCoreTests/HTMLTokenizerTests.swift.

# An event handler is not in any tag's attribute list
input = <p onerror="alert(1)">x</p>
output = <p>x</p>

# Obfuscation inside a dropped attribute; the tags in the content survive
input = <p><span onerror="alert(1)"><em>y</em></span></p>
output = <p><em>y</em></p>

# The scheme allowlist: javascript is not in it
input = <a href="javascript:alert(1)">x</a>
output = x

# A reference in an attribute value is decoded before the URL is read
input = <a href="&#106;avascript:alert(1)">x</a>
output = x

# A control byte in a value with an http prefix: that path never parses
input = <a href="java\nscript:alert(1)">x</a>
output = x

# Both halves of the scheme read through references
input = <a href="&#106;avascript&#58;alert(1)">x</a>
output = x

# data: is not one of the data-URI prefixes for an href
input = <a href="data:text/html,<script>alert(1)</script>">x</a>
output = x

# vbscript is not in the scheme allowlist
input = <a href="vbscript:msgbox(1)">x</a>
output = x

# A hidden element is dropped with its content
input = <div hidden>x</div>
output = (empty)

# A blocked tag is dropped with its content
input = <script>alert(1)</script>
output = (empty)

# A blocked tag inside a table, written at the insertion point
input = <noscript><p>x</p></noscript>
output = (empty)

# td outside a table is not an element at all
input = <table><script>alert(1)</script></table>
output = <table></table>

# img without a usable src has no mandatory attribute
input = <td><script>alert(1)</script></td>
output = (empty)

# srcset: a candidate outside the scheme allowlist, and data: is not allowed
input = <img src="javascript:alert(1)" alt="x">
output = (empty)

# data:text/html is not one of the prefixes the img path allows
input = <img srcset="&#106;avascript:alert(1) 1x, data:text/plain;base64,SGVsbG8= 2x" alt="Example">
output = (empty)

# A prefix in the data-URI allowlist is written as it was read
input = <img src="data:text/html;base64,PHNjcmlwdD5hbGVydCgxKTwvc2NyaXB0Pg==" alt="x">
output = (empty)

# A backslash is an ordinary path character, so the value is a path
input = <img src="data:image/png;base64,AAAA" alt="x">
output = <img src="data:image/png;base64,AAAA" alt="x" loading="lazy">

# The two-solidus value is written as https: plus the value as read
input = <a href="/\\evil.com">x</a>
output = <a href="http://example.org/%5Cevil.com" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# file is not in the scheme allowlist
input = <a href="//evil.example.org/path">x</a>
output = <a href="https://evil.example.org/path" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# iframe is written only for a domain in its allowlist
input = <a href="file:///etc/passwd">x</a>
output = x

# An event handler on a tag in the allowlist is dropped
input = <iframe src="javascript:alert(1)"></iframe>
output = (empty)

# The scheme comparison is case-insensitive
input = <a href="https://evil.com/x" onclick="alert(1)">x</a>
output = <a href="https://evil.com/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# Content inside a kept element keeps the tags in the allowlist
input = <a href="JavaScript:alert(1)">x</a>
output = x

# An attribute value that carries markup is written escaped
input = <p><a href="javascript:alert(1)"><b>x</b></a></p>
output = <p><b>x</b></p>

# An ampersand read as part of an absolute value is escaped by the writer
input = <td rowspan="<b>injection</b>">text</td>
output = text

# Whitespace around a value is not trimmed
input = <a href="https://example.org/a&amp;b">x</a>
output = <a href="https://example.org/a&amp;b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A percent-encoded quote in the path is not decoded by the writer
input = <img src="https://example.org/a&amp;b" alt="x">
output = <img src="https://example.org/a&amp;b" alt="x" loading="lazy">

# The html parse has no depth limit; the guard is the tokenizer's
input = <a href=" https://example.org/x ">x</a>
output = x

# feed is in the scheme allowlist
input = <a href="https://example.org/%22quoted%22">x</a>
output = <a href="https://example.org/%22quoted%22" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A blocked resource substring is checked on the value as it was read
input = <div><div><div><div><div><div><div><div><div><div><div><span>x</span></div></div></div></div></div></div></div></div></div></div></div>
output = x

# A decoded scheme in text is not a URL
input = <p><a href="feed:https://example.org/">x</a></p>
output = <p><a href="feed:https://example.org/" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a></p>

# mailto is in the scheme allowlist
input = <img src="https://stats.wordpress.com/b tracker.gif" alt="x">
output = (empty)

# A space in the path of a value read as absolute is written as read
input = <a href="https://example.org/a b">x</a>
output = <a href="https://example.org/a b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The raw path form is kept when it is not already the canonical encoding
input = <a href="https://example.org/a%2Fb">x</a>
output = <a href="https://example.org/a%2Fb" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The raw form is dropped when it is already the canonical encoding
input = <a href="https://example.org/a%20b">x</a>
output = <a href="https://example.org/a%20b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The query is written as read, so a plus is a plus
input = <a href="https://example.org/?q=a+b">x</a>
output = <a href="https://example.org/?q=a+b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The query is not decoded, so %2B stays %2B
input = <a href="https://example.org/?q=a%2Bb">x</a>
output = <a href="https://example.org/?q=a%2Bb" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An invalid escape is not checked on the as-read path
input = <a href="https://example.org/%zz">x</a>
output = <a href="https://example.org/%zz" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A truncated escape is the same case
input = <a href="https://example.org/a%2">x</a>
output = <a href="https://example.org/a%2" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# Percent escapes in the host are not decoded
input = <a href="http://foo%c3%a9bar/">x</a>
output = <a href="http://foo%c3%a9bar/" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# Tracking parameters are removed through the query round trip
input = <a href="https://example.org/?utm_source=x">x</a>
output = <a href="https://example.org/" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A semicolon inside a query pair is part of the pair
input = <a href="https://example.org/?a=b;c=d">x</a>
output = <a href="https://example.org/?a=b;c=d" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The srcset whitespace class includes 0x0C, so the candidate splits
input = <img srcset="https://example.org/a\x0c 1x,https://example.org/b 2x" alt="x">
output = <img srcset="https://example.org/a 1x, https://example.org/b 2x" alt="x" loading="lazy">

# A decoded tab in a srcset value splits the candidate
input = <img srcset="https://example.org/a&#9; 1x,https://example.org/b 2x" alt="x">
output = <img srcset="https://example.org/a 1x, https://example.org/b 2x" alt="x" loading="lazy">

# Whitespace before the first candidate is skipped by the srcset rule
input = <img srcset="&#9;https://example.org/a 1x" alt="x">
output = <img srcset="https://example.org/a 1x" alt="x" loading="lazy">

# Candidates are written back joined with a comma and one space
input = <img srcset="https://example.org/a 1x , https://example.org/b 2x" alt="x">
output = <img srcset="https://example.org/a 1x, https://example.org/b 2x" alt="x" loading="lazy">

# A decoded colon opens the scheme, and ABC is not in the allowlist
input = <a href="&#65;BC&#58;//example.org/x">x</a>
output = x

# A decoded colon in an allowed scheme still resolves
input = <a href="mailto&#58;a@b.c">x</a>
output = <a href="mailto:a@b.c" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A percent escape in the path is not decoded by the writer
input = <a href="https://example.org/%41">x</a>
output = <a href="https://example.org/%41" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A mixed-case scheme on the as-read path is written as read
input = <a href="HTTPS://example.org/X">x</a>
output = <a href="HTTPS://example.org/X" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# One solidus after the scheme leaves the value opaque
input = <a href="https:/example.org/x">x</a>
output = <a href="https:/example.org/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A protocol-relative value is resolved against the base URL
input = <a href="//example.org/x">x</a>
output = <a href="https://example.org/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The prefix check is case-sensitive, and the parse lower-cases the scheme
input = <a href="HTTP://EXAMPLE.ORG/X">x</a>
output = <a href="HTTP://EXAMPLE.ORG/X" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The port is kept as it was read
input = <a href="https://example.org:8443/x">x</a>
output = <a href="https://example.org:8443/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An invalid port is not checked on the as-read path
input = <a href="https://example.org:abc/x">x</a>
output = <a href="https://example.org:abc/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A bracketed host with no scheme is read as a path
input = <a href="[::1]/x">x</a>
output = x

# The fragment is cut before the scheme, so it is written as read
input = <a href="https://example.org/x#frag">x</a>
output = <a href="https://example.org/x#frag" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A value ending with the query marker keeps the empty query
input = <a href="https://example.org/?#">x</a>
output = <a href="https://example.org/?" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An empty query is written as it was read
input = <a href="https://example.org/?">x</a>
output = <a href="https://example.org/?" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A blocked resource drops the attribute; an a writes only its content
input = <a href="https://twitter.com/share?url=x">x</a>
output = x

# A value starting with a hash is an anchor link: no resolve, no rel
input = <a href="#anchor">x</a>
output = <a href="#anchor">x</a>

# A decoded newline in a value read as absolute is written literally
input = <a href="https://example.org/x&#10;">x</a>
output = <a href="https://example.org/x\n" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A decoded carriage return is written through the writer's escape form
input = <a href="https://example.org/x&#13;">x</a>
output = <a href="https://example.org/x&#13;" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A decoded tab is written literally, not escaped
input = <a href="https://example.org/x&#9;">x</a>
output = <a href="https://example.org/x\t" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A blocked resource drops the attribute, and img writes nothing
input = <img src="https://stats.wordpress.com/b.gif" alt="x">
output = (empty)

# Leading zeros in a numeric reference still resolve
input = <a href="&#0000106;avascript:alert(1)">x</a>
output = x

# A decoded solidus in an img src is not the protocol-relative form
input = <img src="https://example.org/&#x2F;x" alt="x">
output = <img src="https://example.org//x" alt="x" loading="lazy">

# The tracking parameter is removed and the other one is kept
input = <a href="https://example.org/?utm_source=x&keep=1">x</a>
output = <a href="https://example.org/?keep=1" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A scheme in the allowlist inside a kept element, with the a attributes added
input = <p><a href="feed:https://example.org/">x</a></p>
output = <p><a href="feed:https://example.org/" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a></p>

# A tag outside the allowlist writes only its content
input = <p><span hidden>x</span><em>y</em></p>
output = <p><em>y</em></p>

# Cannot pin the order rule: the a in the cell has one attribute
input = <td><a href="https://example.org/x">y</a></td>
output = <a href="https://example.org/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a>

# A relative value goes through the parse, so a space in the path is escaped
input = <a href="/a b">x</a>
output = <a href="http://example.org/a%20b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An invalid escape in the path is a parse failure, so the attribute is skipped
input = <a href="/a%zz">x</a>
output = x

# The query round trip rebuilds the path, so the space is escaped
input = <a href="/a b?utm_source=x">x</a>
output = <a href="http://example.org/a%20b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A decoded control byte in a relative value is a parse failure
input = <a href="/a&#10;b">x</a>
output = x

# A decoded tab in a relative value is a parse failure
input = <a href="/a&#9;b">x</a>
output = x

# A decoded carriage return in a relative value is a parse failure
input = <a href="/a&#13;b">x</a>
output = x

# A control byte in an img src leaves the mandatory attribute missing
input = <img src="/a&#10;b" alt="x">
output = (empty)

# A srcset candidate that fails the parse is dropped, the other is resolved
input = <img srcset="/a&#10;b 1x, /c 2x" alt="x">
output = <img srcset="http://example.org/c 2x" alt="x" loading="lazy">

# A candidate split by whitespace leaves a descriptor that does not parse
input = <img srcset="/a b 1x, /c 2x" alt="x">
output = <img srcset="http://example.org/c 2x" alt="x" loading="lazy">

# The tab between the candidate and its descriptor is the srcset rule
input = <img srcset="/a\t1x">
output = <img srcset="http://example.org/a 1x" loading="lazy">

# A control byte inside a candidate splits it at the whitespace class
input = <img srcset="/a&#9;b 1x">
output = (empty)

# A control byte at the end of a candidate is whitespace for srcset
input = <img srcset="/a&#10; 1x">
output = <img srcset="http://example.org/a 1x" loading="lazy">

# A control byte after the descriptor is whitespace too
input = <img srcset="/a 1x&#10;">
output = <img srcset="http://example.org/a 1x" loading="lazy">

# A doubled percent in a relative path is an escape failure
input = <a href="/a%%b">x</a>
output = x

# The query round trip rebuilds the path but writes the query as read
input = <a href="https://example.org/a b?utm_source=x">x</a>
output = <a href="https://example.org/a%20b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An opaque value is written as it was read, spaces included
input = <a href="mailto:a@b.c ">x</a>
output = <a href="mailto:a@b.c " rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An opaque value with no path is written as it was read
input = <a href="magnet:?xt=urn:btih:abc">x</a>
output = <a href="magnet:?xt=urn:btih:abc" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A control byte in an opaque value is the parse failure
input = <a href="tel:+1&#10;5">x</a>
output = x

# A mixed-case scheme is parsed and written as read because it is absolute
input = <a href="HTTPS://example.org/a b">x</a>
output = <a href="HTTPS://example.org/a b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A relative path already canonical has no raw form to keep
input = <a href="/a%20b">x</a>
output = <a href="http://example.org/a%20b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A relative path keeping its raw form writes it through the rebuild
input = <a href="/a%2Fb">x</a>
output = <a href="http://example.org/a%2Fb" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A protocol-relative value is written as read, so its space is not escaped
input = <a href="//example.org/a b">x</a>
output = <a href="https://example.org/a b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A protocol-relative value is not parsed, so an invalid escape is not an error
input = <a href="//example.org/a%zz">x</a>
output = <a href="https://example.org/a%zz" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A control byte in a query written as read means no cleaning happens
input = <a href="https://example.org/?utm_source=x&#10;">x</a>
output = <a href="https://example.org/?utm_source=x\n" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# An a in a cell is written at the insertion point and resolved
input = <td><a href="/x">y</a></td>
output = <a href="http://example.org/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a>

# A parse failure drops the attribute but not the content
input = <p><a href="/a&#10;b">y</a></p>
output = <p>y</p>

# A data URI is not allowed for a srcset candidate
input = <img srcset="/a 1x, data:image/png;base64,AAAA 2x" alt="x">
output = <img srcset="http://example.org/a 1x" alt="x" loading="lazy">

# The ampersand in a value written as read is escaped, the control byte is not
input = <a href="https://example.org/?utm_source=x&#10;&keep=1">x</a>
output = <a href="https://example.org/?utm_source=x\n&amp;keep=1" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A decoded carriage return in an img src uses the writer's escape form
input = <img src="https://example.org/a&#13;b" alt="x">
output = <img src="https://example.org/a&#13;b" alt="x" loading="lazy">

# Percent escapes in the query are not validated by the parse
input = <a href="/?q=%zz">x</a>
output = <a href="http://example.org/?q=%zz" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The query rebuild writes a decoded space as a plus
input = <a href="/?utm_source=x&q=a%20b">x</a>
output = <a href="http://example.org/?q=a+b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The control check is on the part before the fragment, so a fragment can carry one
input = <a href="https://example.org/x#&#10;">x</a>
output = <a href="https://example.org/x#\n" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A relative path with a space is escaped by the rebuild, the fragment is not
input = <a href="/a b#frag">x</a>
output = <a href="http://example.org/a%20b#frag" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A control byte attached to the descriptor is the srcset whitespace class
input = <img srcset="https://example.org/a\x0c1x, https://example.org/b 2x" alt="x">
output = <img srcset="https://example.org/a 1x, https://example.org/b 2x" alt="x" loading="lazy">

# A mixed-case scheme through the parse and the query round trip is lower-cased
input = <a href="//example.org?a=b">x</a>
output = <a href="https://example.org?a=b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A mixed-case scheme is not the as-read prefix, so its control byte fails the parse
input = <a href="https://example.org/?utm_source=x&keep=1">x</a>
output = <a href="https://example.org/?keep=1" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A named reference without its semicolon is resolved only for a name in the table
input = <a href="a&ampb">x</a>
output = <a href="http://example.org/a&amp;ampb" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# The same lookup in text
input = <p>a&amplt;b</p>
output = <p>a&amp;lt;b</p>

# A name in the table is resolved in text without its semicolon
input = <p>&amp</p>
output = <p>&amp;</p>

# A numeric reference with its semicolon resolves in an attribute value
input = <a href="x&#38;y">x</a>
output = <a href="http://example.org/x&amp;y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A numeric reference without its semicolon stops at the first non-digit
input = <a href="x&#38y">x</a>
output = <a href="http://example.org/x&amp;y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A hex numeric reference resolves in an attribute value
input = <a href="x&#x26;y">x</a>
output = <a href="http://example.org/x&amp;y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# A hex numeric reference without its semicolon resolves too
input = <a href="x&#x26y">x</a>
output = <a href="http://example.org/x&amp;y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">x</a>

# Text is decoded and then escaped again by the writer
input = <pre>&#38;&#60;</pre>
output = <pre>&amp;&lt;</pre>

# A decoded ampersand inside a query pair is part of the pair
input = <p><a href="http://example.org/?utm_source=x&keep=1&#38;">y</a></p>
output = <p><a href="http://example.org/?keep=1" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a></p>

# A blocked tag inside a cell is dropped, and the a after it is written
input = <td><script>alert(1)</script><a href="/x">y</a></td>
output = <a href="http://example.org/x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a>

# A src that fails the parse leaves img without its mandatory attribute
input = <img src="/a%zz" alt="x">x
output = x

# A srcset value that fails the parse leaves source without its attribute
input = <picture><source srcset="/a%zz 1x">fallback</source></picture>
output = <picture>fallback</picture>

# An ampersand in a relative path is decoded and then escaped
input = <a href="/x&#38;y">y</a>
output = <a href="http://example.org/x&amp;y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a>

# A control byte in a non-URL attribute value is written literally
input = <img alt="a&#10;b" src="/x">
output = <img alt="a\nb" src="http://example.org/x" loading="lazy">

# A decoded colon in an href is a scheme, so the attribute is dropped
input = <p><a href="javascript&#58;alert(1)" title="x">y</a></p>
output = <p>y</p>

# Whitespace before the scheme means the prefix check fails
input = <a href=" HTTPS://example.org/x">y</a>
output = y

# A value that is only a decoded ampersand is read as a path and escaped
input = <a href="&amp;">y</a>
output = <a href="http://example.org/&amp;" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">y</a>

# A formatting element writes its attributes in the order the parser sorts them
input = <a title="x" href="/y">z</a>
output = <a href="http://example.org/y" title="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a>

# The sort is by the name, so href comes before id
input = <a id="x" href="/y">z</a>
output = <a href="http://example.org/y" id="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a>

# Three names sorted: href, id, title, whatever the order they were read in
input = <a title="x" id="y" href="/z">q</a>
output = <a href="http://example.org/z" id="y" title="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">q</a>

# The name is lower-cased before the sort
input = <p><a TITLE="x" href="/y">z</a></p>
output = <p><a href="http://example.org/y" title="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a></p>

# The same sort inside a cell, where the token is read by the rules for the body
input = <td><a title="x" href="/y">z</a></td>
output = <a href="http://example.org/y" title="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a>

# A duplicate name is kept once, at the position of the first one
input = <a title="b" href="/y">z</a>
output = <a href="http://example.org/y" title="b" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a>

# abbr is not a formatting element, so its attribute is written as read
input = <abbr title="x">y</abbr>
output = <abbr title="x">y</abbr>

# Cannot pin the sort: the order read is already the sorted order
input = <a href="/y">z</a>
output = <a href="http://example.org/y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a>

# A pixel tracker is dropped before the mandatory-attribute check
input = <img src="https://example.org/x.gif" width="1" height="1" alt="x">
output = (empty)

# One size of 1 is not the pair the pixel-tracker rule needs
input = <img src="https://example.org/x.gif" width="1" alt="x">
output = <img src="https://example.org/x.gif" width="1" alt="x" loading="lazy">

# A trusted iframe domain with the default options, which the reference reads as empty
input = <iframe src="https://www.youtube.com/embed/abc"></iframe>
output = <iframe src="https://www.youtube-nocookie.com/embed/abc" sandbox="allow-scripts allow-same-origin allow-popups allow-popups-to-escape-sandbox" loading="lazy" referrerpolicy="strict-origin-when-cross-origin"></iframe>

# The vimeo rule adds the parameter the reference writes
input = <iframe src="https://player.vimeo.com/video/1"></iframe>
output = <iframe src="https://player.vimeo.com/video/1?dnt=1" sandbox="allow-scripts allow-same-origin allow-popups allow-popups-to-escape-sandbox" loading="lazy"></iframe>

# The sort applies to a formatting element inside another formatting element
input = <b><a title="x" href="/y">z</a></b>
output = <b><a href="http://example.org/y" title="x" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">z</a></b>

# A decoded character in text is written as the character
input = <a href="/y">&#106;</a>
output = <a href="http://example.org/y" rel="noopener noreferrer" referrerpolicy="no-referrer" target="_blank">j</a>

# The apostrophe uses the writer's escape form
input = <img src="https://example.org/x" alt="a&#39;b">
output = <img src="https://example.org/x" alt="a&#39;b" loading="lazy">
